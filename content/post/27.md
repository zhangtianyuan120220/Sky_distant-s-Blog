## 题目理解

- 通过题目给的公式可以进一步推出 $awa_p \gets a_0 \oplus a_1 \oplus a_2 \dots \oplus a_{p-1}$（其中 $awa$ 表示一个前缀异或数组）。

- 用前缀异或的方式快速得出的公式带进去得到一个前缀异或数组。

- 找到符合要求的答案，并将它转化成二进制。

::::warning
但此时还需要进位和去除前导零。
::::

## 代码解决
```cpp
#include<bits/stdc++.h>
using namespace std;
void solve(){
    int n;
    cin>>n;
    vector<int> a(n+10);
    vector<int> awa(n+10,0);//前缀异或数组 
    vector<int> ans(n+10,0);//最终答案数组 
    for(int i=0;i<=n;i++){
        cin>>a[i];
    }
    //套公式 awa[i] = a[0]^...^a[i-1]
    for(int i=0;i<=n;i++){
        awa[i+1]=awa[i]^a[i];
    }
    for(int i=0;i<n;i++){//得出结果 
        if(awa[i]==a[i]){
            ans[n-i-1]++;
        }
    }
    if(awa[n]==a[n]){
        ans[0]++;
    }
    //进位
    for(int i=0;i<=n+3;i++){
        if(ans[i]>=2){
            ans[i+1]+=ans[i]>>1;
            ans[i]&=1;
        }
    }
    //找最高位
    int len=n+4;
    while(len>=0&&ans[len]==0){//删除前导零 
		len--;
	}
    if(len<0){//判断如果全是零，就输出0 
    	puts("0");
        return;
    }
    for(int i=len;i>=0;i--){//输出答案 
        cout<<ans[i];
    }
    puts("");
}
int main() {
    int t;
    cin>>t;
    while(t--){
        solve();
    }
    return 0;
}//Sky_distant
