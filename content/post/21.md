## 题目分析
- 本题需要判断是否存在一种数组排列方式，使得对于任意分割点 $i$，前缀的 $\operatorname{MEX}$ 值与后缀的 $\operatorname{MEX}$ 值不同。
- 经过分析，可以得到一个简单的判断条件：统计每个数的出现频率，如果存在一个数 $v$，它至少出现 `2` 次，并且 $v+1$ 也至少出现 `1` 次，那么答案是 `NO`，否则是 `YES`。

## 代码解决
```cpp
#include<bits/stdc++.h>
using namespace std;
int t;
inline void solve(){//定义一个整数n，用于存储序列的长度
    int n;//序列长度变量
    cin>>n;//输入序列长度
    vector<int> a(n);//创建一个大小为n的向量a，用于存储序列元素
    map<int,int> awa;//创建一个映射awa，用于记录每个数字出现的次数
    for(int i=0;i<n;i++){
        cin>>a[i];//输入序列元素
        awa[a[i]]=awa[a[i]]+1;//统计每个数字出现的次数
    }
    bool ans=true;//初始化答案为true，表示序列有效
    map<int,int>::iterator it;//创建映射迭代器
    for(it=awa.begin();it!=awa.end();it++){
        int f=it->first;//获取当前数字
        int s=it->second;//获取当前数字出现的次数
        if(s>=2){//如果数字出现次数大于等于2
            if(awa.find(f+1)!=awa.end()){//检查是否存在相邻数字
                ans=false;//如果存在相邻数字，则设置答案为false
                break;//提前终止循环
            }
        }
    }
    if(ans){//根据答案输出结果
        puts("YES");
    }
    else {
        puts("NO");
    }
}
int main(){
    cin>>t;
    while(t--){
        solve();
    }
    return 0;
