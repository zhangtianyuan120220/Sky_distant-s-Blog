## 题目分析
- 将每个员工的每个月的工资存入一个二维数组中，在算应纳税款。
- 有两种收入，一种是 `PAY`,另一种是 `INCOME`。
- 当收入方式 为 `PAY` 时，每月结算一次，需将这个月的所有 `PAY` 收入累加，最终在以总工资减去 800 元的金额根据题目缴税。
- 当收入方式为 `INCOME` 时，有一次就要缴一次税，不超过 4000 元，就以减去 800 元的金额缴税，超过 4000 元，就以减除百分之二十的金额缴税。

## 代码实现
```cpp
#include<bits/stdc++.h>
using namespace std;
double a[50005][13],ans=0;
double fw[]={0,500,2000,5000,20000,40000,60000,80000,100000};
double sl[]={0.05,0.10,0.15,0.20,0.25,0.30,0.35,0.40,0.45};
int m;
string awa;
double pay(double qian){
    double now=qian-800;
    double ans=0;
    if(now<=0){
        return 0;
    }
    for(int i=0;i<8;i++){
        if(now>fw[i+1]){
            ans+=(fw[i+1]-fw[i])*sl[i];
        }
        else{
            ans+=(now-fw[i])*sl[i];
            return ans;
        }
    }
    ans+=(now-100000)*0.45;
    return ans;
}
double income(double qian){
    double now,ans=0;
    if(qian<=4000)now=qian-800;
    else now=qian*0.8;
    if(now<=0)return 0;
    if(now<=20000)ans=now*0.2;
    else if(now<=50000)ans=20000*0.2+(now-20000)*0.3;
    else ans=20000*0.2+30000*0.3+(now-50000)*0.4;
    return ans;
}
int main(){
    cin>>m;
    while(cin>>awa&&awa!="#"){
        int name,month,day;
        double qian;
        scanf("%d %d/%d %lf",&name,&month,&day,&qian);
        if(awa=="PAY"){
            a[name][month]+=qian;
        }
        if(awa=="INCOME"){
            ans+=income(qian);
        }
    }
    for(int i=1;i<=m;i++) {
        for(int j=1;j<=12;j++){
            if(a[i][j]>0){
                ans+=pay(a[i][j]);
            }
        }
    }
    cout<<fixed<<setprecision(2)<<ans<<endl;
    return 0;
}//Sky_distant
