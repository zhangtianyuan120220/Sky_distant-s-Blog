## 题目分析
- 输入一个 $R$ 行 $C$ 列的网格表示地图，其中 `.` 表示海洋，`X` 表示陆地。要求将三面或四面临海的陆地变为海洋 **（地图外被认为是海洋）**，输出最终的包含所有陆地的最小矩形。

### 具体做法
- 遍历一遍整个地图，当是 `X` 时，搜索他的**上下左右**，并用一个变量计算他的临海面数，当大于等于三时，就将他标记上。
- 最后将整个地图缩减成包含所有陆地的最小矩阵。

## 代码解决
```cpp
#include<bits/stdc++.h>
using namespace std;
int r,c;
int xx[6]={0,0,1,0,-1,0};
int yy[6]={0,1,0,-1,0,0};
char a[3000][3000];
int ans=0;
int x=0,y=0,z=0,q=0;
void sx(){//搜索出包含说有陆地的最小矩阵 
	for(int i=1;i<=r;i++){
		for(int j=1;j<=c;j++){
			if(a[i][j]=='X'){
				x=min(x,i);
				z=max(z,i);
				y=min(y,j);
				q=max(q,j);				
			}
		}
	}
}
void sc(){//输出结果 
	for(int i=x;i<=z;i++){
		for(int j=y;j<=q;j++){
			cout<<a[i][j];
		}
		puts("");
	}	
}
void xc(){//将原先标记的 # 还原为 .  
	for(int i=1;i<=r;i++){
		for(int j=1;j<=c;j++){
			if(a[i][j]=='#')a[i][j]='.';
		}
	}	
}
int main(){
	cin>>r>>c;
	x=r,y=c;
	for(int i=1;i<=r;i++){
		for(int j=1;j<=c;j++){
			cin>>a[i][j];
		}
	}
	for(int i=1;i<=r;i++){
		for(int j=1;j<=c;j++){
			if(a[i][j]=='X'){
				//判断当前陆地有几面临海 
				int qwq=0;
				for(int x=1;x<=4;x++){
					if(a[i+xx[x]][j+yy[x]]=='.')qwq++;
					if(i+xx[x]<=0||j+yy[x]<=0||i+xx[x]>r||j+yy[x]>c)qwq++;
				}
				if(qwq>=3){
					a[i][j]='#';//被淹没的陆地被标记为 #  
				}
			}
		}
	}
	xc();//还原矩阵 
	sx();//找到最小矩阵 
	sc();//输出 
	return 0;
}//Sky_distant
