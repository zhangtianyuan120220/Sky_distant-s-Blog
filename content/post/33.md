## 题目分析
- 可以将第一个字符串认为是明码，第二个字符串认为是暗码，通过两个字符串中的字母出现频率映射，进行解码。
- 可以先使用结构体来存储两个字符串中字符出现的频率及其字母，之后再将数据扔进 `vector` 中排序，最后用 `map` 映射将第二个字符串解码，得到最终答案。

::::warning[注意]
最后答案的输出要注意，除了最后一组答案只用换一行，其余要换两行，否则会爆零。~~（别问我怎么知道的）~~
::::

## 代码实现
```cpp
#include<bits/stdc++.h>
using namespace std;
struct Node{//结构体用来存储两个字符串中字母出现的频率
    int num;
    char zm;
};
bool cmp(Node a, Node b){//比较函数
    if(a.num != b.num) return a.num > b.num;
    return a.zm < b.zm;
}
int t;
int main(){
//	freopen("Key_to_Success.in","r",stdin);
//	freopen("Key_to_Success.out","w",stdout);
// 文件测评记得写
    cin >> t;
    while(t--){
        Node a[256], b[256];
        string awa,qwq;//两个字符串
        string ans = "";//最终答案
        cin >> awa >> qwq;
        for(int i = 0; i < 256; i++){//初始化结构体
            a[i].num = 0;
            a[i].zm = char(i);
            b[i].num = 0;
            b[i].zm = char(i);
        }
        for(int i = 0; i < awa.size(); i++){
            char c = awa[i];
            a[c].num++;
            a[c].zm = c;
        }//存储awa中字母出现频率
        for(int i = 0; i < qwq.size(); i++){
            char c = qwq[i];
            b[c].num++;
            b[c].zm = c;
        }//存储qwq中字母出现频率
        vector<Node> ans1, ans2;//使用vector节省空间
        for(int i = 0; i < 256; i++){
            if(a[i].num > 0){
            	ans1.push_back(a[i]);
			}
            if(b[i].num > 0){
            	ans2.push_back(b[i]);
			} 
        }
        sort(ans1.begin(), ans1.end(), cmp);//排序
        sort(ans2.begin(), ans2.end(), cmp);//排序
        map<char, char> mp;
        for(int i = 0; i < min(ans1.size(), ans2.size()); i++){
            mp[ans2[i].zm] = ans1[i].zm;//映射出答案
        }
        for(int i = 0; i < qwq.size(); i++){
            char c = qwq[i];
            ans += mp[c];//记录答案
        }
        if(t==0){//注意最后答案只换一次行
            cout << ans;
			puts("");
        }
        else{
        	cout<<ans;
        	puts("");
        	puts("");
  		}
    }
    return 0;
}
