## 题目分析
- 输入一个起始日期和一个结束日期（格式为 `YYYYMMDD`），问这个时间段内有几个特殊日期，特殊日期就是在日期中至少有三个连续的相同数字。
- 直接暴力枚举所有日期，在一个个判断是否为特殊日期。
- 在枚举日期是还要判断当前年份是否是闰年，当前月份有几天。

## 代码实现
```cpp
#include<bits/stdc++.h>
using namespace std;
int days[]={0,31,28,31,30,31,30,31,31,30,31,30,31};
int start,end;
int ans=0;
int a,b;
bool rn(int y){
    return (y%4==0&&y%100!=0)||(y%400==0);
}
bool check(int num){
    char s[10];
    sprintf(s,"%08d",num);//格式化为 8 位带前导零的字符串
    for(int i=0;i<=5;i++){
        if(s[i]==s[i+1]&&s[i+1]==s[i+2])return 1;
    }
    return 0;
}
int main(){
    cin>>start>>end;
    a=start/10000;
    b=end/10000;
    for(int y=a;y<=b;y++){
    	if(rn(y)){
    		days[2]=29;
		}
		else days[2]=28;
        for(int m=1;m<=12;m++){
            for(int d=1;d<=days[m];d++){
                int now=y*10000+m*100+d;
                if(now>=start&&now<=end){
                    if(check(now)){
                        ans++;
                    }
                }
            }
        }
	}
    cout<<ans<<endl;
    return 0;
}//Sky_distant
